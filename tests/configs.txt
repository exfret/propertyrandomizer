# Settings configs for dev/run-tests.py; every config runs on every mod set in tests/mod-configs
# Setting names leave out the "propertyrandomizer-" prefix
#
# Most configs are generated from settings.lua, so new settings are covered without touching this file
# Covered settings are the ones players can see, plus the hidden unified-* ones; other hidden settings are unfinished and left out
# A setting also needs a finite set of values to be covered: a bool, a list of allowed values, or a minimum and a maximum
#   smoke:    defaults, and every covered visible setting at its last value (true for bools, the maximum for numbers)
#   settings: each non-default value of each covered visible setting on its own, then pairwise combinations (every two values of two settings meet in some config)
#   unified:  each non-default value of each unified-* setting on its own, all unified-* bools on over several seeds, and everything on at once
# Plus the hand-picked configs at the bottom

# wip SUITE: the suite is still being worked on, so its failures are reported but don't fail the run
wip unified

# tier NAME PATTERN...: a small named set of runs for dev/run-tests.py --tier NAME; a run is in it if its name contains any pattern
# precommit runs from dev/git-hooks/pre-commit, and should take about 10 minutes
tier precommit /defaults /max /unit-tests base/single-technology= base/single-recipe= base/single-recipe-tech-unlock= base/single-item=

# skip SETTING: leave a setting out of generated configs
skip seed                   # each run picks its own seed
skip tech-tree-rebuild      # unfinished: nothing reads it yet
# Being worked on (visible for now only for that)
skip planetary-oceans
skip planetary-resources
skip unified-entity
skip unified-entity-biters

# requires SETTING OTHER=VALUE: SETTING does nothing unless OTHER=VALUE, so generated configs that change SETTING set OTHER too
requires item-percent item=true
requires recipe-science-pack recipe=true
requires unified-entity-biters unified-entity=true

# config NAME SUITE [seeds=A,B,...] [nocheck=CHECK,...] SETTING=VALUE...: a hand-picked config (no spaces inside a value)
# nocheck turns off the reachability, mechcheck or control check (see dev/run-tests.py)
# In-game unit tests (tests/execute.lua); they stop before randomization, so there's no reachability data or MECHCHECK verdict
config unit-tests smoke nocheck=reachability,mechcheck test-unit=true
config graph-standard settings technology=true recipe=true item=true recipe-tech-unlock=true
# Override syntax is documented in OVERRIDES.md; the first two are its own examples
config overrides-toggles settings overrides=belt_speed;!crafting_machine_speed
config overrides-set settings overrides=RESET:[entity=inserter].NUMERICAL;SET:[entity=inserter].rotation_speed=(X+Y)/2
# Malformed overrides should only add warnings
config overrides-malformed settings overrides=RESET:inserter;SET:[entity=inserter];NOT_AN_INSTRUCTION:x;not_a_randomization
