# Settings configs for dev/run-tests.py; every config runs on every mod set in tests/mod-configs, unless a limit line below narrows a mod set
# Setting names leave out the "propertyrandomizer-" prefix
#
# Most configs are generated from settings.lua, so new settings are covered without touching this file
# Covered settings are the ones players can see, plus the hidden unified-* ones; other hidden settings are unfinished and left out
# A setting also needs a finite set of values to be covered: a bool, a list of allowed values, or a minimum and a maximum
#   smoke:    defaults, and every covered visible setting at its last value (true for bools, the maximum for numbers)
#   settings: each non-default value of each covered visible setting on its own, then pairwise combinations (every two values of two settings meet in some config)
#   unified:  each non-default value of each unified-* setting on its own, and all unified-* bools on over several seeds (never with other settings: randomizations still in development are only tested in isolation)
# Plus the hand-picked configs at the bottom

# wip SUITE: the suite is still being worked on, so its failures are reported but don't fail the run
wip unified

# tier NAME PATTERN...: a small named set of runs for dev/run-tests.py --tier NAME; a run is in it if its name contains any pattern
# precommit runs from dev/git-hooks/pre-commit, and should take about 10 minutes
tier precommit base/defaults sa/defaults base/max sa/max base/unit-tests sa/unit-tests base/single-technology= base/single-recipe= base/single-recipe-tech-unlock= base/single-item=

# limit MODSET CONFIG...: mod sets matching MODSET only run the configs named (both can use *, like unified-all@*); naming a mod set with --modset runs every config on it
# Any Planet Start (the aps-* mod sets, each starting on another planet through its aps-planet setting) is checked on the default randomizations, everything players can turn on, and the graph randomizations
# Run just these with: dev/run-tests.py --only aps-
limit aps-* defaults max graph-standard

# oldlogic SETTING=VALUE: configs with this setting are judged by the old logic's check of the built game (OLDLOGICCHECK in data-final-fixes.lua) instead of the reachability and MECHCHECK checks
# TEMPORARY (user, 2026-09-28): old item randomization only has to pass against the old logic for now, so it doesn't block the next release
oldlogic item=true

# skip SETTING: leave a setting out of generated configs
skip seed                   # each run picks its own seed
skip tech-tree-rebuild      # unfinished: nothing reads it yet
# The unified preview turns off the old graph randomizations, which generated configs like max are there to test, so it gets its own configs below
skip unified-preview
# Being worked on (visible for now only for that)
skip planetary-oceans
skip planetary-resources
skip planetary-lightning
skip planetary-freezing
skip planetary-locks
skip unified-entity
skip unified-entity-biters

# requires SETTING OTHER=VALUE: SETTING does nothing unless OTHER=VALUE, so generated configs that change SETTING set OTHER too
requires item-percent item=true
requires recipe-science-pack recipe=true
requires unified-entity-biters unified-entity=true

# pin SETTING=VALUE NAME...: configs in one of these suites or with one of these names (generated or hand-picked) get SETTING=VALUE on top of their own settings
# Development handlers (dev-unified, on by default in the checkout and off in releases) are only tested in the unified suite
pin dev-unified=false smoke settings
pin dev-unified=true unified
pin dev-unified=false preview preview-with-graph

# config NAME SUITE [seeds=A,B,...] [nocheck=CHECK,...] SETTING=VALUE...: a hand-picked config (no spaces inside a value)
# nocheck turns off reachability, mechcheck, control or science-cost (see dev/run-tests.py)
# In-game unit tests (tests/execute.lua); they stop before randomization, so there's no reachability data or MECHCHECK verdict
config unit-tests smoke nocheck=reachability,mechcheck test-unit=true
# The unified preview as releases have it (dev-unified off), so the preview alone has to turn on the unified randomizations still in development
config preview unified seeds=1,2,3,4 unified-preview=true dev-unified=false
# With old graph randomizations on too, which the preview turns off (not item: its configs are judged by the old logic's check, which then doesn't run)
# No science cost check either: it measures old recipe randomization, which is off here
config preview-with-graph unified nocheck=science-cost unified-preview=true dev-unified=false technology=true recipe=true recipe-science-pack=true recipe-tech-unlock=true
config graph-standard settings technology=true recipe=true item=true recipe-tech-unlock=true
# Global science cost ceilings, before/after recipes: each pack <=4x, arithmetic mean of ratios <2x.
# Run: python3 dev/run-tests.py settings --modset base --modset sa --only science-cost
# Missing baseline or final prices fail: every science pack must be covered.
config science-cost settings seeds=749,750,751 recipe=true
# Override syntax is documented in OVERRIDES.md; the first two are its own examples
config overrides-toggles settings overrides=belt_speed;!crafting_machine_speed
config overrides-set settings overrides=RESET:[entity=inserter].NUMERICAL;SET:[entity=inserter].rotation_speed=(X+Y)/2
# Malformed overrides should only add warnings
config overrides-malformed settings overrides=RESET:inserter;SET:[entity=inserter];NOT_AN_INSTRUCTION:x;not_a_randomization
